    function analyzeName(name) {

      const chars = name.toLowerCase().split('');

      let freqSum = 0;

      chars.forEach((c, i) => {

        const code = c.charCodeAt(0);

        freqSum += (code - 96) * (i + 1);

      });

      const baseFreq = 200 + (freqSum % 400);


      const vowels = 'aeiou';

      let vowelCount = 0;

      chars.forEach(c => { if (vowels.includes(c)) vowelCount++; });

      

      const symmetry = Math.max(3, Math.min(12, chars.length * 2 - vowelCount + 3));

      const uniqueChars = new Set(chars).size;

      const harmonics = Math.max(2, Math.min(8, uniqueChars));


      const patternTypes = ['RADIAL', 'SPIRAL', 'CRYSTAL', 'FLUID', 'GEOMETRIC'];

      const patternIndex = Math.floor(freqSum % patternTypes.length);


      const waveComponents = [];

      chars.forEach((c, i) => {

        const code = c.charCodeAt(0);

        waveComponents.push({

          frequency: 1 + (code % 10) * 0.5,

          amplitude: 0.3 + (code % 7) * 0.1,

          phase: (i / chars.length) * Math.PI * 2

        });

      });


      return { baseFreq, symmetry, harmonics, pattern: patternTypes[patternIndex], waveComponents, charCount: chars.length };

    }


    function initParticles() {

      particles = [];

        particles.push({

          x: centerX + Math.cos(angle) * radius,

          y: centerY + Math.sin(angle) * radius,

          homeX: centerX + Math.cos(angle) * radius,

          homeY: centerY + Math.sin(angle) * radius,

          vx: 0, vy: 0,

          size: 1 + Math.random() * 1.5

        });

      }

    }


    function getNodalAmplitude(x, y, params, t) {

      const centerX = canvasWidth / 2;

      const centerY = canvasHeight / 2;

      const maxR = Math.min(canvasWidth, canvasHeight) * 0.45;

      

      const nx = (x - centerX) / maxR;

      const ny = (y - centerY) / maxR;

      const r = Math.sqrt(nx * nx + ny * ny);

      const theta = Math.atan2(ny, nx);

      

      let amplitude = 0;

      const m = params.symmetry;

      const n = params.harmonics;

      

      const radialNode = Math.cos(r * n * Math.PI + t * 0.5);

      

      let angularNode = 0;

      for (let k = 1; k <= m; k++) { angularNode += Math.cos(k * theta * 2 + t * 0.3 * k); }

      angularNode /= m;

      

      amplitude = radialNode * angularNode;

      

      params.waveComponents.forEach((wave) => {

        const dist = Math.sqrt(nx * nx + ny * ny);

        amplitude += Math.sin(dist * wave.frequency * 10 + t * wave.frequency + wave.phase) * wave.amplitude * 0.3;

      });

      

      return amplitude;

    }


    function updateParticles(params, t) {

      const centerX = canvasWidth / 2;

      const centerY = canvasHeight / 2;

      const maxR = Math.min(canvasWidth, canvasHeight) * 0.45;

      

      particles.forEach(p => {

        const amplitude = getNodalAmplitude(p.x, p.y, params, t);

        const epsilon = 2;

        const ampX1 = getNodalAmplitude(p.x + epsilon, p.y, params, t);

        const ampX2 = getNodalAmplitude(p.x - epsilon, p.y, params, t);

        const ampY1 = getNodalAmplitude(p.x, p.y + epsilon, params, t);

        const ampY2 = getNodalAmplitude(p.x, p.y - epsilon, params, t);

        

        const gradX = (ampX1 - ampX2) / (2 * epsilon);

        const gradY = (ampY1 - ampY2) / (2 * epsilon);

        

        const forceStrength = 0.5;

        p.vx -= gradX * forceStrength + (p.x - p.homeX) * 0.001;

        p.vy -= gradY * forceStrength + (p.y - p.homeY) * 0.001;

        

        const jitter = Math.abs(amplitude) * 2;

        p.vx += (Math.random() - 0.5) * jitter;

        p.vy += (Math.random() - 0.5) * jitter;

        

        p.vx *= 0.9; p.vy *= 0.9;

        p.x += p.vx; p.y += p.vy;

        

        const dx = p.x - centerX;

        const dy = p.y - centerY;

        const dist = Math.sqrt(dx * dx + dy * dy);

        

        if (dist > maxR) {

          const angle = Math.atan2(dy, dx);

          p.x = centerX + Math.cos(angle) * maxR;

          p.y = centerY + Math.sin(angle) * maxR;

          p.vx *= -0.5; p.vy *= -0.5;

        }

      });

    }

    }

    // MAIN GENERATION (Triggered ONLY by Button)

    // ============================================

    

    function generatePattern(name) {

      if (!name.trim()) return;

      

      currentParams = analyzeName(name);

      

      freqDisplay.textContent = `${currentParams.baseFreq} Hz`;

      symmetryVal.textContent = `${currentParams.symmetry}-fold`;

      harmonicsVal.textContent = currentParams.harmonics;

      patternVal.textContent = currentParams.pattern;

      

      patternInfo.classList.remove('hidden');

      initParticles();

      time = 0;

      startVibration();

      

      setTimeout(() => { stopVibration(); }, 4000);

    }


    // ============================================

    // EVENT LISTENERS

    // ============================================

    

    generateBtn.addEventListener('click', () => { generatePattern(nameInput.value); });

    nameInput.addEventListener('keypress', (e) => { if (e.key === 'Enter') { generatePattern(nameInput.value); } });


    // ============================================

    // INITIALIZE

    // ============================================

    

    function drawIdle() {

      ctx.fillStyle = '#0a0a0a';

      ctx.fillRect(0, 0, canvasWidth, canvasHeight);

      ctx.strokeStyle = 'rgba(255, 255, 255, 0.03)';

      ctx.lineWidth = 0.5;

      const gridSize = 20;

      for (let x = 0; x < canvasWidth; x += gridSize) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, canvasHeight); ctx.stroke(); }

      for (let y = 0; y < canvasHeight; y += gridSize) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvasWidth, y); ctx.stroke(); }

      ctx.fillStyle = 'rgba(255, 255, 255, 0.2)';

      ctx.font = '12px "IBM Plex Mono", monospace';

      ctx.textAlign = 'center';

      ctx.fillText('ENTER NAME TO BEGIN', canvasWidth / 2, canvasHeight / 2);

    }


    animate();

    drawIdle();
